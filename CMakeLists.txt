# Development flags belong to project-owned targets, not downstream consumers.
function(gig_project_target target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
    if(GIG_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
    if(GIG_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
  set(sanitizers "")
  if(GIG_ENABLE_ASAN)
    list(APPEND sanitizers address)
  endif()
  if(GIG_ENABLE_UBSAN)
    list(APPEND sanitizers undefined)
  endif()
  if(GIG_ENABLE_TSAN)
    list(APPEND sanitizers thread)
  endif()
  if(sanitizers)
    list(JOIN sanitizers "," sanitizer_flags)
    target_compile_options(${target} PRIVATE -fsanitize=${sanitizer_flags} -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(${target} PRIVATE -fsanitize=${sanitizer_flags} -fno-sanitize-recover=all)
  endif()
endfunction()

option(GIG_BUILD_PYTHON "Build optional pinned pybind11 research interface" OFF)
if(GIG_BUILD_PYTHON)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Research wire bindings require a 64-bit target")
  endif()
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter Development.Module)
  if(Python3_VERSION VERSION_GREATER_EQUAL "3.13")
    message(FATAL_ERROR "Pinned research bindings currently support CPython 3.9 through 3.12")
  endif()
  set(PYBIND11_FINDPYTHON ON)
  find_package(pybind11 2.13.6 EXACT CONFIG REQUIRED)
  set_target_properties(gig_microstructure PROPERTIES POSITION_INDEPENDENT_CODE ON)
  pybind11_add_module(_native MODULE NO_EXTRAS python/bindings/module.cpp python/bindings/wire.cpp python/bindings/schemas.cpp)
  target_link_libraries(_native PRIVATE gig::microstructure)
  gig_project_target(_native)
  string(TOUPPER "${CMAKE_BUILD_TYPE}" GIG_BUILD_TYPE_UPPER)
  set(GIG_CONFIG_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${GIG_BUILD_TYPE_UPPER}}")
  get_target_property(gig_native_options _native COMPILE_OPTIONS)
  get_target_property(gig_core_options gig_microstructure COMPILE_OPTIONS)
  string(JOIN " " GIG_NATIVE_OPTIONS ${gig_native_options})
  string(JOIN " " GIG_CORE_OPTIONS ${gig_core_options})
  configure_file(python/bindings/build_info.hpp.in "${CMAKE_BINARY_DIR}/generated/gig_python_build_info.hpp" @ONLY)
  target_include_directories(_native PRIVATE "${CMAKE_BINARY_DIR}/generated")
  set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python/gig_microstructure")
  file(GLOB python_sources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/python/gig_microstructure/*.py")
  foreach(source ${python_sources})
    get_filename_component(name "${source}" NAME)
    configure_file("${source}" "${CMAKE_BINARY_DIR}/python/gig_microstructure/${name}" COPYONLY)
